program PROG;
var int16_t A, B, SUM, I, MAX, RES

start
    %% 1. Ініціалізація та граничні значення (до 32767) %%
    A <- 10;
    B <- 32000;
    MAX <- 32767;

    %% 2. Ввід та Вивід (get, put) %%
    get SUM;
    put MAX;

    %% 3. Арифметика (+, -, *, /, %) %%
    SUM <- A + B * 2 / 5;
    RES <- SUM % 10;

    %% 4. Логіка (eg, ne, and, or, !, >=, <=) %%
    if A ne 0 and B >= A or ! SUM eg 0 start
        A <- A - 1
    finish
    else start
        A <- 0
    finish;

    %% 5. Цикл з діапазоном (..) %%
    for I in 1 .. 10 start
        SUM <- SUM + I
    finish
finish

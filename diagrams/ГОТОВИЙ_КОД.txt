  1️⃣  ЗАГАЛЬНА РОБОТА ТРАНСЛЯТОРА
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@startuml
title Загальна робота транслятора
skinparam defaultTextAlignment center

start

:Читання файлу .z07;

:Лексичний аналіз
(Lexer.tokenize);

:Отримання токенів;

:Синтаксичний аналіз
(Parser.parse);

if (Синтаксис OK?) then (так)
  :Семантичний аналіз
  (Semantic.analyze);

  if (Семантика OK?) then (так)
    :Генерація C коду
    (CodeGen.generate);

    :Збереження output.c;

    :Компіляція gcc;

    if (Компіляція OK?) then (так)
      :Готово!
      Executable: ./output;
      stop
    else (ні)
      :Помилка компіляції;
      stop
    endif
  else (ні)
    :Семантичні помилки;
    stop
  endif
else (ні)
  :Синтаксичні помилки;
  stop
endif

@enduml

  2️⃣  ЛЕКСИЧНИЙ АНАЛІЗАТОР
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@startuml
title Лексичний аналізатор (Lexer)
skinparam defaultTextAlignment center

start

:pos = 0, line = 1;

while (pos < len?) is (так)

  if (пробіл?) then (так)
    :pos++, якщо newline то line++;

  elseif (коментар %%?) then (так)
    :Пропустити до %%
    skipComment();

  elseif (цифра?) then (так)
    :readNumber();
    if (> 32767?) then (так)
      :ERROR;
    else (ні)
      :Token NUMBER;
    endif

  elseif (літера?) then (так)
    :readWord();

    if (ключове слово?) then (так)
      :Token K_XXX;
    elseif (оператор слово?) then (так)
      :Token OP_XXX;
    else (ні)
      if (Up4 правило OK?) then (так)
        :Token IDENTIFIER;
      else (ні)
        :ERROR;
      endif
    endif

  else (оператор)
    :Token OP_XXX
    (; , + - * / etc);
  endif

endwhile (ні)

:Token END_OF_FILE;

stop

@enduml

  3️⃣  СИНТАКСИЧНИЙ АНАЛІЗАТОР
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@startuml
title Синтаксичний аналізатор (Parser)
skinparam defaultTextAlignment center

start

:pos = 0;

:parseProgram();

:match(PROGRAM);
:match(IDENTIFIER);
:match(SEMICOLON);

:match(VAR);
:parseVariableDeclaration();

:match(START);

while (current != FINISH?) is (так)

  if (IDENTIFIER?) then (так)
    :parseAssignment()
    ID <- expr;

  elseif (GET?) then (так)
    :parseInput()
    get ID;

  elseif (PUT?) then (так)
    :parseOutput()
    put expr;

  elseif (IF?) then (так)
    :parseIf();

  elseif (FOR?) then (так)
    :parseFor();

  else (помилка)
    :ERROR;
    stop
  endif

endwhile (ні)

:match(FINISH);

:Успіх!;

stop

@enduml

  4️⃣  ГЕНЕРАТОР КОДУ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@startuml
title Генератор коду (CodeGenerator)
skinparam defaultTextAlignment center

start

:Записати заголовок C;

:generateProgram();

:Записати "int main() {";

:Оголосити змінні
int16_t VAR1, VAR2, ...;

while (current != FINISH?) is (так)

  if (IDENTIFIER?) then (так)
    :Згенерувати присвоєння
    varName = expr;;

  elseif (GET?) then (так)
    :Згенерувати введення
    scanf("%hd", &var);;

  elseif (PUT?) then (так)
    :Згенерувати вивід
    printf("%d", expr);;

  elseif (IF?) then (так)
    :Згенерувати if
    if (cond) { ... };

  elseif (FOR?) then (так)
    :Згенерувати for
    for (i=start; i<=end; i++);

  endif

endwhile (ні)

:Записати "return 0;";
:Записати "}";

:Зберегти в output.c;

stop

@enduml

  5️⃣  СЕМАНТИЧНИЙ АНАЛІЗАТОР
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

@startuml
title Семантичний аналізатор (Semantic)
skinparam defaultTextAlignment center

start

:Створити таблицю символів;

:analyzeProgram();

:Оголошення змінних
додати в symbolTable;

while (current != FINISH?) is (так)

  if (IDENTIFIER присвоєння?) then (так)
    if (змінна оголошена?) then (так)
      :OK, позначити
      ініціалізованою;
    else (ні)
      :ERROR: not declared;
    endif

  elseif (GET?) then (так)
    if (змінна оголошена?) then (так)
      :OK, позначити
      ініціалізованою;
    else (ні)
      :ERROR: not declared;
    endif

  elseif (PUT?) then (так)
    :Перевірити вираз;

  elseif (IF?) then (так)
    :Перевірити умову та тіло;

  elseif (FOR?) then (так)
    :Перевірити змінну циклу;

  endif

endwhile (ні)

if (є помилки?) then (так)
  :Вивести помилки;
  stop
else (ні)
  :Вивести таблицю символів;
  :Успіх!;
  stop
endif

@enduml
